/* ============ Utils ============ */

const clamp=n=>Math.max(0,Math.min(1,n));

function avg(objs){const k=Object.keys(objs[0]||{}),o={};for(const x of k)o[x]=objs.reduce((s,v)=>s+v[x],0)/objs.length;return o;}

function cosine(a,b){const K=Object.keys(a),d=K.reduce((s,k)=>s+a[k]*b[k],0),na=Math.sqrt(K.reduce((s,k)=>s+a[k]*a[k],0)),nb=Math.sqrt(K.reduce((s,k)=>s+b[k]*b[k],0));return d/(na*nb||1);}

const sleep=ms=>new Promise(r=>setTimeout(r,ms));


/* ============ Feature Model ============ */

const BASE_GENRE={

  "Hip-hop":{energy:.7,sophistication:.55,groove:.85,structure:.5,experiment:.6,urban:.8,global:.6},

  "K-Pop":{energy:.85,sophistication:.8,groove:.65,structure:.9,experiment:.55,urban:.65,global:.95},

  "World":{energy:.7,sophistication:.45,groove:.85,structure:.45,experiment:.6,urban:.65,global:.9},

/* ============ Open Web Adapters ============ */

async function fetchITunesArtists(term,limit=10){

  const url=`https://itunes.apple.com/search?term=${encodeURIComponent(term)}&entity=musicArtist&limit=${limit}`;

  const r=await fetch(url); const j=await r.json();

  return (j.results||[]).map(x=>({name:x.artistName,genres:[mapGenre(x.primaryGenreName)]}));

}

async function fetchMBArtists(term,limit=8){

  const url=`https://musicbrainz.org/ws/2/artist?query=${encodeURIComponent(term)}&limit=${limit}&fmt=json`;

  const r=await fetch(url); if(!r.ok) return [];

  const j=await r.json();

  const base=(j.artists||[]).map(a=>({name:a.name,id:a.id}));

  const out=[];

  for(const a of base){

    try{

      const t=await fetch(`https://musicbrainz.org/ws/2/artist/${a.id}?inc=tags&fmt=json`);

      const tj=await t.json(); const tags=(tj.tags||[]).map(z=>z.name);

      out.push({name:a.name,genres:tagsToGenres(tags)}); await sleep(110);

    }catch{ out.push({name:a.name,genres:[]}); }

  }

  return out;

}

function tagsToGenres(tags){const txt=(tags||[]).join(" ").toLowerCase(); for(const [re,g] of GENRE_MAP){if(re.test(txt)) return [g];} return [];}

function mapGenre(s){if(!s) return "Indie"; for(const [re,g] of GENRE_MAP){if(re.test(String(s))) return g;} return "Indie";}

async function buildOpenCatalog(keys){

  const map=new Map();

  const add=arr=>arr.forEach(a=>{const k=a.name.toLowerCase(); map.set(k,{name:a.name,genres:(a.genres&&a.genres.length?a.genres:["Indie"])})});

  for(const k of keys){ add(await fetchITunesArtists(k,12).catch(()=>[])); add(await fetchMBArtists(k,8).catch(()=>[])); }

  if(map.size<12){ add(await fetchITunesArtists("electronic",10).catch(()=>[])); add(await fetchITunesArtists("funk",10).catch(()=>[])); }

  return [...map.values()];

}

    {label:"実験・創造推進力",score:clamp(0.7*v.experiment+0.2*v.global)},

    {label:"推進・行動エネルギー",score:v.energy},

    {label:"越境・多文化統合力",score:clamp(0.7*v.global+0.2*v.experiment)}

  ];

}

function detectType(v){

  if(v.experiment>0.65&&v.global>0.6) return "Curious Innovator";

  if(v.groove>0.7&&v.urban>0.6) return "Empathic Connector";

  if(v.structure>0.7&&v.sophistication>0.6) return "System Architect";

  if(v.energy>0.7&&v.groove>0.6) return "Dynamic Performer";

  return "Serene Observer";

}

    default:return "静けさを味方に。お気に入りの曲を背景にリズムを整えよう。";

  }

}


/* ============ Recommendation ============ */

function vecFromArtist(a){return toVec(a.genres&&a.genres.length?a.genres:["Indie"]);}

function recommend(userVec, have, catalog){

  const pool=catalog.filter(a=>!have.has(a.name.toLowerCase()));

  pool.forEach(a=>{

    const v=vecFromArtist(a);

    const sim=cosine(userVec,v);

    const growth=0.3*Math.max(0,v.experiment-userVec.experiment)+0.2*Math.max(0,v.global-userVec.global)+0.2*Math.max(0,v.urban-userVec.urban);

    a._score=sim+growth;

  });

  const top=pool.sort((a,b)=>b._score-a._score);

  const picked=[], used=new Set();

  for(const a of top){

    const g=(a.genres&&a.genres[0])||"Indie"; const penalty=used.has(g)?0.08:0;

    if(picked.length<10 && a._score-penalty>0.55){picked.push(a); used.add(g);}

  }

  return picked;

}

}
